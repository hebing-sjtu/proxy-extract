# Pinned to the versions this pipeline was actually developed and measured on.
# Loosen deliberately, not by accident: numpy 2.x and transformers 5.x are both
# breaking-change majors, and the semantic backend reads `id2label` off the
# checkpoint config, which transformers has moved before.
#
# `scripts/setup_venv.sh` installs from this file, so it is the single record of
# the versions the measurements in RUNBOOK.md were taken on.

# --- core, needed for contract / taxonomy / QC / encoding (no GPU) ---
numpy==2.5.1
pillow==12.3.0
opencv-python-headless==5.0.0.93
# Static ffmpeg build. The delivery encoders shell out to ffmpeg, which cannot
# be pip-installed any other way, so this is what lets a venv work on a node
# where nobody has root. A system ffmpeg on PATH still wins over it.
imageio-ffmpeg==0.6.0

# --- model backends ---
# On linux/x86_64 the default PyPI torch wheel already bundles CUDA, so no
# extra index is needed — but which CUDA it bundles is PyPI's choice, and for
# this version it is 13.0, which needs a 580+ driver. CUDA is only compatible
# across minor versions, so on a node whose driver is 12.x that wheel cannot
# see the GPUs at all: torch.cuda.is_available() is False and everything falls
# back to the CPU behind a single UserWarning. Same versions, built for 12.6:
#
#   pip install --index-url https://download.pytorch.org/whl/cu126 \
#       torch==2.13.0+cu126 torchvision==0.28.0+cu126
#
# The `+cu126` matters: plain `torch==2.13.0` is already satisfied by the cu130
# build, so pip would do nothing. run_scenes.sh refuses to launch if this is
# wrong; `nvidia-smi --query-gpu=driver_version --format=csv` says which it is.
torch==2.13.0
torchvision==0.28.0
transformers==5.16.1
accelerate==1.14.0
safetensors==0.8.0
huggingface-hub==1.29.0

# --- tests ---
pytest==9.1.1
